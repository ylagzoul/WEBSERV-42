events {
    worker_connections 1024;
}

http {

    server {
        listen 127.0.0.1:8000;
        listen 127.0.0.1;
        server_name mysite;

        root ./www;
        index index.html;

        error_page 404 /errors/404.html;

        location / {
            listen hello;
            server_name mysite;
        }

        listen 1.1.1.1;


        location /images {
            root ./www/images;
            methods GET;
        }
    }

}
