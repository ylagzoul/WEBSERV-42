server {
    listen 127.0.0.1:8080;

    root /var/www/main;
    server_name = a..com b.com c.com

    error_page 404 /errors/404.html;
    error_page 500 /errors/500.html;

    location / {
        methods GET POST;
        root /var/www/main;
        index index.html;
        autoindex off;
    }

    location /images {
        methods GET;
        root /var/www/images;
        autoindex on;
    }

    location /blog {
        methods GET;
        root /var/www/blog;
        index index.html;
        autoindex off;
    }

    location /upload {
        methods POST;
        root /var/www/uploads;
        upload on;
    }
}
